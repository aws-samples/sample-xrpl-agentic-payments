# XRPL Agentic Payments MCP Server - Runtime Dependencies
#
# This is the single dependency manifest for the MCP server program
# (src/mcp_server/server.py), for BOTH of the artifacts that ship it:
#   * the AgentCore Runtime zip built by deploy/package.sh (deps pre-installed)
#   * the container image built by the repo-root Dockerfile
# Keeping one manifest per program is what stops the two artifacts from
# drifting apart the way a hand-maintained `pip install a b c` list did.

# XRPL SDK (blockchain interaction)
xrpl-py==5.0.0

# MCP server framework (FastMCP for streamable-http).
# Capped below 2.0 on purpose: mcp 2.x moved FastMCP from
# `mcp.server.fastmcp` to `mcp.server.mcpserver`, so an unbounded pin resolves
# to a release where `from mcp.server.fastmcp import FastMCP` — the first
# import in server.py — raises ModuleNotFoundError at startup.
mcp[cli]>=1.9.0,<2

# Async compatibility
nest-asyncio>=1.6.0

# Environment variable loading
python-dotenv>=1.0.0

# Transitive dependency pins (security fixes)
# python-multipart: CVE-2024-53981, CVE-2026-24486, CVE-2026-42561 fixed in 0.0.27+
python-multipart>=0.0.31
# httpx: not imported by any module in this repo (leftover from a removed
# OpenSanctions HTTP call). Installed transitively by xrpl-py and mcp, so the
# line is kept as a version FLOOR only — deleting it would drop the >=0.27.0
# security baseline on a package that ships either way.
httpx>=0.27.0
